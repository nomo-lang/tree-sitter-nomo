package sample

import std.task
import std.time

interface Runner {
    suspend fn run(self)
}

pub suspend fn main() {
    task.scope {
        task.deadline(time.duration_millis(5)) {
            task.yield_now()
        }
        task.select {
            task.receive(messages) => message {
                consume(message)
            }
            task.sleep(time.duration_millis(50)) => timeout {
                consume(timeout)
            }
        }
    }
}
